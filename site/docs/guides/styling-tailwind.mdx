---
title: Styling with Tailwind
description: Style the tree with Tailwind's data and ARIA variants, arbitrary values for the depth, and one class string per part kept next to the component.
---

The state attributes map one to one onto Tailwind's variants: `data-selected:`,
`data-expanded:`, `data-leaf:`, `data-[drop-position=inside]:`, `aria-disabled:`. The depth is an
arbitrary value over `--tree-depth`. Nothing in the packages assumes Tailwind; the examples use
it, with one class string per part in a `styles.ts` next to each component.

## The parts

```ts title="styles.ts"
export const styles = {
    root: "overflow-auto rounded-lg border p-1 text-sm",
    item: "outline-none",
    row: [
        "flex h-8 cursor-default select-none items-center gap-1.5 rounded-md pe-2",
        "ps-[calc(var(--tree-depth)*1rem+0.25rem)] hover:bg-gray-100",
        "[:focus-visible>&]:ring-2 [:focus-visible>&]:ring-blue-600 [:focus-visible>&]:ring-inset",
        "data-selected:bg-blue-100 data-selected:text-blue-900 data-disabled:opacity-50",
    ].join(" "),
    toggle: [
        "flex size-4 shrink-0 items-center justify-center transition-[rotate] data-leaf:invisible",
        "data-expanded:rotate-90 rtl:-scale-x-100 rtl:data-expanded:-rotate-90",
    ].join(" "),
    label: "min-w-0 flex-1 truncate",
};
```

<Framework name="react">

```tsx
import { type ItemOf, Tree } from "@fragiola/tree-react";
import { ChevronRight } from "lucide-react";
import { styles } from "./styles";

function FileItem({ item }: { item: ItemOf<Types> }) {
    return (
        <Tree.Item item={item} className={styles.item}>
            <Tree.ItemContent className={styles.row}>
                <Tree.Toggle className={styles.toggle}>
                    <ChevronRight aria-hidden className="size-3.5" />
                </Tree.Toggle>
                <span className={styles.label}>{item.label}</span>
            </Tree.ItemContent>
            <Tree.Group<Types>>{(child) => <FileItem item={child} />}</Tree.Group>
        </Tree.Item>
    );
}
```

</Framework>

<Framework name="angular">

```ts title="file-tree.ts"
import { Component } from "@angular/core";
import { TREE } from "@fragiola/tree-angular";
import { LucideChevronRight } from "@lucide/angular";
import { styles } from "./styles";
import { json } from "./tree";

@Component({
    selector: "app-file-tree",
    imports: [TREE, LucideChevronRight],
    template: `
        <div treeRoot [defaultTree]="json" aria-label="Files" [class]="styles.root">
            <ng-template [treeItems]="json" let-item>
                <div [treeItem]="item" [class]="styles.item">
                    <div treeItemContent [class]="styles.row">
                        <span treeToggle [class]="styles.toggle">
                            <svg lucideChevronRight aria-hidden="true" class="size-3.5"></svg>
                        </span>
                        <span [class]="styles.label">{{ item.label }}</span>
                    </div>
                    <div treeGroup></div>
                </div>
            </ng-template>
        </div>
    `,
})
export class FileTree {
    protected readonly json = json;
    protected readonly styles = styles;
}
```

</Framework>

- `[:focus-visible>&]:` draws the ring on the row when its item (the element with DOM focus)
  shows focus.
- `ps-[calc(var(--tree-depth)*1rem+0.25rem)]` indents by depth, mirrored in right-to-left.
- `data-leaf:invisible` keeps a leaf's toggle room, so labels line up.

## State as a function

<Framework name="react">

`className` also takes a function of the part's state, for the cases a variant cannot express:

```tsx
<Tree.ItemContent className={(state) => cn(styles.row, state.level === 1 && "font-semibold")}>
    {item.label}
</Tree.ItemContent>
```

</Framework>

<Framework name="vue">

A Vue `class` has no function form. Most states are a variant away (`data-[depth=0]:font-semibold`
for the root level); for the rest, the `render` slot's `state`:

```vue
<Tree.ItemContent :class="styles.row">
    <template #render="{ props, state, children }">
        <div v-bind="props" :class="{ 'font-semibold': state.level === 1 }">
            <component :is="children" />
        </div>
    </template>
    {{ item.label }}
</Tree.ItemContent>
```

</Framework>

<Framework name="angular">

In Angular, `class` has no function form. Most states are a variant away
(`data-[depth=0]:font-semibold` for the root level); for the rest, bind on the directive's
`state()`:

```html
<div treeItemContent #row="treeItemContent" [class]="styles.row" [class.font-semibold]="row.state().level === 1">
    {{ item.label }}
</div>
```

</Framework>

## Themes

The examples read their sizes and colours from CSS variables (`--tr-indent`, `--tr-row-height`,
`--tr-selected-bg`, …) set by a theme file, so one set of class strings works in the light and
the dark theme: `ps-[calc(var(--tree-depth)*var(--tr-indent)+0.25rem)]`,
`data-selected:bg-(--tr-selected-bg)`. Copy that pattern to theme your own tree.
